# The stock Rufus writes a stick and the stick has one letter (docs/disks.md).
# As r3-rufus-write.txt, with letters.exe copied onto the same disk as rufus.exe
# (two disks come up in any order on a first start). The pauses are for the
# emulated processor (--tcg), where a dialog comes up after seconds.
#   mcopy -i out/test-local/rufus-apps.img@@1M out/tests/letters.exe ::letters.exe
#   vm-test.py --tcg --disk out/test-local/rufus-apps.img --script tests/vm/l3-rufus-one-letter.txt
# Expected: l3-before and l3-after show the stick on the same one letter (E:).
sleep 8
# the first Run box of a start is slow to take keys
key win-r
sleep 20
key esc
sleep 5
plug target rufus-stick.img SERIALRUFUS005
sleep 15

sleep 15
type cmd /k D:\letters.exe

sleep 15

key ret

click
