# Arch packages the host image is assembled from. This is a source of files,
# not a distribution: nothing here is visible to the user.

# Base
filesystem
glibc
gcc-libs
tzdata
kmod

# Hardware manager: only udevd, udevadm, sysusers and hwdb are used
systemd

# Names of PCI devices: dwm.exe names the display adapter from pci.ids
hwdata

# Serial shell in development builds; its udhcpc and ip configure wired networks
busybox

# Networking: Wi-Fi association and WPA (iwd, driven over the system bus by
# the Windows side), the regulatory database, root certificates for TLS
iwd
dbus
wireless-regdb
ca-certificates

# Libraries the Wine unix side loads
freetype2
gnutls
alsa-lib
libusb

# Sound: alsactl init brings every card's mixer to sensible levels (the
# kernel leaves them muted); alsa-lib brings the UCM profiles along
alsa-utils
vulkan-icd-loader
vulkan-swrast
wayland
libxkbcommon
libinput

# GPU drivers' user space: OpenGL (radeonsi, iris, crocus, nouveau) and Vulkan
# (radv, anv, NVK) with GBM and EGL. NVIDIA's own driver is installed apart,
# at the version of its kernel modules (kernel/NVIDIA_VERSION)
mesa
vulkan-radeon
vulkan-intel
vulkan-nouveau

# GPU and network firmware (the drivers come with the kernel)
linux-firmware-amdgpu
linux-firmware-radeon
linux-firmware-intel
linux-firmware-nvidia
linux-firmware-realtek
linux-firmware-mediatek
linux-firmware-atheros
linux-firmware-broadcom
